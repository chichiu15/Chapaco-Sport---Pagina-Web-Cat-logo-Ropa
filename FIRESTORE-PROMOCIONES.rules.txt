match /packs_ofertas/{packId} {
  // El cliente necesita leer los packs activos. La interfaz filtra estado y vigencia.
  allow read: if true;

  // El panel administrativo ya exige inicio de sesión y permiso de promociones.
  allow create, update, delete: if request.auth != null;
}

match /codigos_descuento/{codigoId} {
  // El cliente solo consulta un código exacto; no puede descargar la lista completa.
  allow get: if true;
  allow list: if request.auth != null;
  allow create, update, delete: if request.auth != null;
}
